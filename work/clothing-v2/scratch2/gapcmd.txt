import numpy as np, builtins, bpy
from mathutils import Vector
hero, body = builtins._HCACHE['MatchHero_Male']
import shoe2
fr = shoe2.Frame(body, 'R')
ob = [o for o in bpy.data.objects if o.name.startswith('Kit_Shoe')][0]
me = ob.data; n = len(me.vertices); p = np.zeros(n * 3); me.vertices.foreach_get('co', p); p = p.reshape(-1, 3)
panel = np.zeros(len(me.polygons), np.int32); me.attributes['panel'].data.foreach_get('value', panel)
mat = np.zeros(len(me.polygons), np.int32)
for i, pl in enumerate(me.polygons): mat[i] = pl.material_index
vs = sorted({v for pl, pa, mi in zip(me.polygons, panel, mat) if pa == 0 and mi == 0 for v in pl.vertices})
q = p[vs]
# foot-only vertices as a point cloud (z>0.0, weights foot)
d = np.array([body.bvh.find_nearest(Vector(v))[3] for v in q])
fq = fr.to(q)
L = fr.L
for a in np.linspace(0, 1, 11)[:-1]:
    m = (fq[:, 1] / L >= a) & (fq[:, 1] / L < a + .1) & (fq[:, 2] > .03)
    if m.sum(): print('GAP y/L %.1f-%.1f  n=%d  median %.1f mm  p90 %.1f mm  max %.1f mm' % (a, a + .1, m.sum(), np.median(d[m]) * 1e3, np.percentile(d[m], 90) * 1e3, d[m].max() * 1e3))
