        sealed class Seg { public float len; public Action begin; public Action<float> tick; public Action end; }
        Camera cam; Action beforeCapture;
        IEnumerator Play(string name, List<Seg> segs)
        {
            string dir = Path.Combine(Root, name); Directory.CreateDirectory(dir);
            foreach (var f in Directory.GetFiles(dir, "*.jpg")) File.Delete(f);
            int frame = 0;
            foreach (var seg in segs)
            {
                seg.begin?.Invoke();
                int n = Mathf.Max(1, Mathf.RoundToInt(seg.len * FPS));
                for (int i = 0; i < n; i++)
                {
                    yield return new WaitForEndOfFrame();
                    cam = Camera.main; cam.aspect = (float)W / H;
                    seg.tick?.Invoke(i / (float)FPS);
                    beforeCapture?.Invoke();
                    Canvas.ForceUpdateCanvases();
                    GameCapture.Save(Path.Combine(dir, $"f{frame++:0000}.jpg"), W, H);
                }
                seg.end?.Invoke();
            }
            Clear();
        }
        void Cam(Vector3 position, Vector3 look, float verticalFov)
        {
            cam.transform.position = position; cam.transform.rotation = Quaternion.LookRotation(look - position, Vector3.up);
            cam.fieldOfView = verticalFov;
        }
        static float S(float x) { x = Mathf.Clamp01(x); return x * x * (3 - 2 * x); }
        static float S2(float x) { x = Mathf.Clamp01(x); return x * x * x * (x * (x * 6 - 15) + 10); }
        static float Out(float x) { x = Mathf.Clamp01(x); return 1 - (1 - x) * (1 - x) * (1 - x); }
        static float Spring(float x) { x = Mathf.Clamp01(x); return 1 - Mathf.Exp(-x * 7) * Mathf.Cos(x * 11); }

        void Slide(RectTransform node, float t, float len, bool fromRight = false, float travel = 900)
        {
            float enter = Out(t / .45f), exit = S((t - (len - .4f)) / .4f);
            float side = fromRight ? 1 : -1;
            var p = node.anchoredPosition; float home = (fromRight ? -64 : 64);
            node.anchoredPosition = new Vector2(home + side * travel * (1 - enter) + side * travel * exit, p.y);
            var g = node.GetComponent<CanvasGroup>(); if (g) g.alpha = Mathf.Min(enter * 1.5f, 1 - exit);
        }

        static void Fade(RectTransform node, float t, float len, float ramp = .3f)
        {

            node.GetComponent<CanvasGroup>().alpha = a; node.localScale = Vector3.one * Mathf.Lerp(.92f, 1, Out(t / ramp));
        }

        readonly Dictionary<RectTransform, bool> ribbonFrom = new();
        void Slam(RectTransform node, float t, float len, float baseY)
        {
            bool fromRight = ribbonFrom.TryGetValue(node, out var r) && r;
            float side = fromRight ? 1 : -1;
            float enter = Spring(t / .5f), exit = S((t - (len - .3f)) / .3f);
            var dir = (Vector2)(node.localRotation * Vector3.right);
            // in from its side, past to the other side on the way out
            node.anchoredPosition = new Vector2(0, baseY) + dir * (side * 2600 * (1 - enter) - side * 2600 * exit);
            node.GetComponent<CanvasGroup>().alpha = 1 - exit;
        }

        RawImage SpeedLines(Color c)
        {
            if (!linesTex)
            {
                const int N = 512; linesTex = new Texture2D(N, N, TextureFormat.RGBA32, false);
                var px = new Color32[N * N]; var rng = new System.Random(4);
                var rays = new float[160]; for (int i = 0; i < rays.Length; i++) rays[i] = (float)(rng.NextDouble() * Math.PI * 2);
                for (int y = 0; y < N; y++) for (int x = 0; x < N; x++)
                {
                    float dx = x - N / 2f, dy = y - N / 2f, r = Mathf.Sqrt(dx * dx + dy * dy) / (N / 2f), a = Mathf.Atan2(dy, dx);
                    float best = 0; foreach (var ray in rays) { float d = Mathf.Abs(Mathf.DeltaAngle(a * Mathf.Rad2Deg, ray * Mathf.Rad2Deg)); best = Mathf.Max(best, 1 - d / .9f); }
                    float alpha = Mathf.Clamp01(best) * Mathf.Clamp01((r - .38f) / .25f);
                    px[y * N + x] = new Color32(255, 255, 255, (byte)(alpha * 255));
                }
                linesTex.SetPixels32(px); linesTex.Apply();
            }
            var root = Node("Speed lines", ui, new Vector2(.5f, .5f), new Vector2(.5f, .5f), Vector2.zero, new Vector2(2600, 2600));
            lines = new GameObject("Lines").AddComponent<RawImage>(); lines.transform.SetParent(root, false); lines.texture = linesTex; lines.color = c; lines.raycastTarget = false;
            var rt = lines.rectTransform; rt.anchorMin = Vector2.zero; rt.anchorMax = Vector2.one; rt.offsetMin = rt.offsetMax = Vector2.zero;
            return lines;
        }

        // ---------------------------------------------------------------- golf
        GolfGame golf; GolferView golfer; Hole hole; Vector3 tee, pin, feet, facing, sideOut, lineDir; int sea = 1;
        readonly List<Vector3> path = new(); readonly List<float> dist = new(); float length;
        IEnumerator SetUpGolf(int jumpTo = 0)
        {
            if (!golf || SceneManager.GetActiveScene().name != "Golf") { yield return SceneManager.LoadSceneAsync("Golf"); yield return null; golf = Object.FindFirstObjectByType<GolfGame>(); }
            beforeCapture = null;
            golf.enabled = true;
            var r0 = Object.FindFirstObjectByType<CameraRig>(); if (r0) r0.enabled = true;
            golf.EquipResultEmotes(new[] { "bringIt", "wave", "pushups" });
            golf.PrepareNativeAddress(Environment.GetEnvironmentVariable("INTRO_MOCK_COURSE") ?? "cliffside"); golf.NativeReady(); golf.TryNativePresentation();
            yield return null; yield return null;
            if (jumpTo > 0) { golf.JumpToHole(jumpTo); yield return null; yield return null; }
            golf.enabled = false;
            var rig = Object.FindFirstObjectByType<CameraRig>(); if (rig) { rig.RestoreFov(); rig.enabled = false; }
            var hud = (Hud)typeof(GolfGame).GetField("hud", Private).GetValue(golf);
            hud.HideHoleIntro(); hud.HideNameplate(); hud.ShowPlayHud(false);
            hole = golf.CurrentHole;
            golfer = (GolferView)typeof(GolfGame).GetField("golfer", Private).GetValue(golf);
            double h = hole.Tee.HeadingTo(hole.RecommendedTarget(hole.Tee)) * Math.PI / 180;
            var aim = new Vector3((float)Math.Sin(h), 0, (float)Math.Cos(h));
            golfer.SetVisible(true); golfer.Stand(golf.BallPosition, aim); golfer.Perform("Idle"); lineDir = aim;
            feet = golfer.transform.position; facing = golfer.transform.forward; facing.y = 0; facing.Normalize();
            tee = HoleView.ToWorld(hole.Tee); pin = HoleView.ToWorld(hole.Pin);
            path.Clear(); dist.Clear(); length = 0;
            var pts = new List<Vector3>(); foreach (var p in hole.Centerline) pts.Add(HoleView.ToWorld(p));
            Vector3 last = pts[0];
            for (int i = 0; i <= 120; i++)
            {
                var p = Catmull(pts, i / 120f); p.y = G(p);
                if (i > 0) { var d = p - last; d.y = 0; length += d.magnitude; }
                path.Add(p); dist.Add(length); last = p;
            }
            float left = 0, right = 0;
            for (float u = .3f; u <= .8f; u += .1f)
            {
                var at = At(u); var r = Right(u);
                for (float d = 15; d <= 150; d += 15) { if (Water(at + r * d)) { right += 150 - d; break; } }
                for (float d = 15; d <= 150; d += 15) { if (Water(at - r * d)) { left += 150 - d; break; } }
            }
            sea = right >= left ? 1 : -1;
            sideOut = Vector3.Cross(Vector3.up, facing);
            MakeCanvas();
        }
        bool Water(Vector3 p) { var lie = hole.LieAt(new CoursePoint(p.x, p.z)); return lie is CourseLie.Water or CourseLie.OutOfBounds; }
        static float G(Vector3 p) => (float)HoleView.GroundHeight(HoleView.ToCourse(p));
        static Vector3 Above(Vector3 p, float clear) { p.y = Mathf.Max(p.y, G(p) + clear); return p; }
        static Vector3 Catmull(List<Vector3> pts, float u)
        {
            int n = pts.Count - 1; float s = Mathf.Clamp01(u) * n; int i = Mathf.Min((int)s, n - 1); float f = s - i;
            Vector3 P(int k) => pts[Mathf.Clamp(k, 0, n)];
            Vector3 p0 = P(i - 1), p1 = P(i), p2 = P(i + 1), p3 = P(i + 2);
            return .5f * ((2 * p1) + (-p0 + p2) * f + (2 * p0 - 5 * p1 + 4 * p2 - p3) * f * f + (-p0 + 3 * p1 - 3 * p2 + p3) * f * f * f);
        }
        Vector3 OnPath(float u)
        {
            float want = Mathf.Clamp01(u) * length; int n = 1;
            while (n < dist.Count - 1 && dist[n] < want) n++;
            return Vector3.Lerp(path[n - 1], path[n], Mathf.InverseLerp(dist[n - 1], dist[n], want));
        }
        Vector3 Fwd(float u) { var a = OnPath(u + .04f) - OnPath(u - .04f); a.y = 0; return a.sqrMagnitude > 1e-4f ? a.normalized : facing; }
        Vector3 At(float u) => u < 0 ? path[0] - Fwd(0) * (-u * length) : u > 1 ? path[^1] + Fwd(1) * ((u - 1) * length) : OnPath(u);
        Vector3 Right(float u) => Vector3.Cross(Vector3.up, Fwd(u));
        string HoleLine => $"HOLE {hole.PlayNumber}  ·  {hole.Name.ToUpperInvariant()}  ·  PAR {hole.Par}  ·  {hole.Length:F0} YD";
        string Event => "CLIFFSIDE OPEN";
        void Emote(string move) { if (!golfer.HasMove(move) || !golfer.Perform(move, loop: false)) golfer.Perform("Wave"); }
        /// Front of the golfer (the intro emotes face this way), turned `degrees` round, `back` yards out.
        Vector3 Front(float degrees, float back, float up) => feet + Quaternion.AngleAxis(degrees, Vector3.up) * facing * back + Vector3.up * up;

        RawImage Shade(RectTransform parent, bool right)
        {
            if (!shadeTex)
            {
                const int N = 256; shadeTex = new Texture2D(N, N, TextureFormat.RGBA32, false); shadeTex.wrapMode = TextureWrapMode.Clamp;
                var px = new Color32[N * N];
                for (int y = 0; y < N; y++) for (int x = 0; x < N; x++)
                {
                    float dx = x / (float)N, dy = y / (float)N; float r = Mathf.Sqrt(dx * dx * .55f + dy * dy * 1.3f);
                    float a = Mathf.Clamp01(1 - r) ; a = a * a * .62f;
                    px[y * N + x] = new Color32(4, 12, 28, (byte)(a * 255));
                }

        List<Feature> Features()
        {
            var all = new List<Feature>();
            var P = new Func<Vector3, CoursePoint>(v => new CoursePoint(v.x, v.z));
            // a forced carry over water or lava on the line of play
            float wetFrom = -1, wetTo = -1, longestFrom = 0, longest = 0;
            for (float d = 4; d < length - 4; d += 2)
            {
                bool wet = hole.LieAt(P(AtYards(d))) == CourseLie.Water;
                if (wet && wetFrom < 0) wetFrom = d;
                if (!wet && wetFrom >= 0) { wetTo = d; if (wetTo - wetFrom > longest) { longest = wetTo - wetFrom; longestFrom = wetFrom; } wetFrom = -1; }
            }
            if (longest > 15)
            {
                var mid = AtYards(longestFrom + longest / 2); bool lava = hole.HazardAt(P(mid)) == HazardKind.Lava;
                all.Add(new Feature { kind = "carry", label = lava ? "LAVA CARRY" : "WATER CARRY", sub = $"{longest:F0} YD TO CLEAR", at = mid, size = longest, score = 100 });
            }
            // the windmill's sails
            if (hole.Windmill != null)
                all.Add(new Feature { kind = "windmill", label = "THE WINDMILL", sub = "TIME THE SAILS", at = new Vector3((float)hole.Windmill.HubX, (float)hole.Windmill.HubY, (float)hole.Windmill.HubD), size = (float)hole.Windmill.Reach, score = 95 });
            // hazards near the line of play: greenside and fairway bunker groups, water and lava pools
            var greenside = new List<Vector3>(); var fairway = new List<Vector3>(); float fairwayAlong = 0;
            foreach (var h in hole.Hazards)
            {
                var c = new Vector3((float)h.X, 0, (float)h.Distance); c.y = G(c);
                float toPin = Vector2.Distance(new Vector2(c.x, c.z), new Vector2(pin.x, pin.z)); float off = (float)hole.DistanceFromCenterline(P(c)); float along = AlongOf(c);
                if (h.Kind == HazardKind.Bunker)
                {
                    if (toPin < hole.GreenRadius + 34) greenside.Add(c);
                    else if (off < hole.FairwayWidth / 2 + 22 && along > 110) { fairway.Add(c); fairwayAlong = fairway.Count == 1 ? along : Mathf.Min(fairwayAlong, along); }
                }
                else if (h.Kind is HazardKind.Water or HazardKind.Lava && off < h.Width / 2 + 30)
                    all.Add(new Feature { kind = "pool", label = h.Kind == HazardKind.Lava ? "LAVA" : "WATER", sub = $"{along:F0} YD · {(Vector3.Dot(c - AtYards(along), Right(along / length)) > 0 ? "RIGHT" : "LEFT")}", at = c, size = (float)Math.Max(h.Width, h.Length), score = 80 });
            }
            // (the one in the way: nearest the line of play)
            Vector3 Nearest(List<Vector3> v) { var best = v[0]; double bd = double.MaxValue; foreach (var x in v) { double d = hole.DistanceFromCenterline(P(x)); if (d < bd) { bd = d; best = x; } } return best; }
            if (greenside.Count > 0) all.Add(new Feature { kind = "greenside", label = greenside.Count > 1 ? "GREENSIDE BUNKERS" : "GREENSIDE BUNKER", sub = greenside.Count > 1 ? $"{greenside.Count} GUARD THE GREEN" : "GUARDS THE GREEN", at = Nearest(greenside), size = 18, score = 55 + 5 * greenside.Count });
            if (fairway.Count > 0) all.Add(new Feature { kind = "fairway", label = fairway.Count > 1 ? "FAIRWAY BUNKERS" : "FAIRWAY BUNKER", sub = $"FROM {fairwayAlong:F0} YD", at = Nearest(fairway), size = 22, score = 50 + 5 * fairway.Count });
            // rocks and walls standing in play
            int rocks = 0; var rockAt = Vector3.zero;
            foreach (var o in hole.Obstacles)
                if (o.Kind is ObstacleKind.Rock or ObstacleKind.Wall && hole.DistanceFromCenterline(new CoursePoint(o.X, o.D)) < hole.FairwayWidth / 2 + 8) { rocks++; rockAt += new Vector3((float)o.X, (float)o.Base, (float)o.D); }
            if (rocks > 0) all.Add(new Feature { kind = "rocks", label = "ROCKS IN PLAY", sub = $"{rocks} ON THE LINE", at = rockAt / rocks, size = 12, score = 45 });
            // the sea close beside the fairway, when nothing has to be carried
            if (longest <= 15)
            {
                float closest = float.MaxValue; Vector3 edge = Vector3.zero; float edgeAlong = 0; int edgeSide = 1;
                for (float d = length * .35f; d < length * .95f; d += 8)
                    foreach (int side in new[] { -1, 1 })
                        for (float o = 10; o < 60; o += 5)
                        {
                            var q = AtYards(d) + Right(d / length) * side * o;
                            if (hole.LieAt(P(q)) == CourseLie.Water) { if (o < closest) { closest = o; edge = q; edgeAlong = d; edgeSide = side; } break; }
                        }
                if (closest < 50) { edge.y = G(AtYards(edgeAlong)); all.Add(new Feature { kind = "edge", label = hole.SeaIsLava ? "LAVA EDGE" : "CLIFF EDGE", sub = $"{(hole.SeaIsLava ? "LAVA" : "THE SEA")} {(edgeSide > 0 ? "RIGHT" : "LEFT")} · {edgeAlong:F0} YD", at = edge, size = 30, score = 45 }); }
            }
            all.Sort((a, b) => b.score.CompareTo(a.score));
            var picked = new List<Feature>();
            foreach (var f in all)
            {
                if (picked.Count == 2) break;
                bool near = Vector3.Distance(f.at, pin) < 22 && f.kind != "greenside";
                foreach (var g in picked) near |= Vector3.Distance(f.at, g.at) < 40;
                if (!near) picked.Add(f);
            }
            picked.Sort((a, b) => AlongOf(a.at).CompareTo(AlongOf(b.at)));
            float rise = G(pin) - G(tee);
            picked.Add(new Feature { kind = "green", label = "THE GREEN", sub = rise > 4 ? $"UPHILL · +{rise:F0} YD" : rise < -4 ? $"DOWNHILL · {rise:F0} YD" : $"PAR {hole.Par} · {hole.Length:F0} YD", at = pin, size = (float)hole.GreenRadius, score = 0 });
            Debug.Log("[IntroMockup] hole " + hole.Number + " features: " + string.Join(", ", all.ConvertAll(f => f.label + " " + f.score)) + " -> " + string.Join(", ", picked.ConvertAll(f => f.label)));
            return picked;
        }
        /// Of the directions round `subject`, the first (nearest the preferred one) from which the lens is in the clear all
        /// the way through its push and nothing stands between it and what it looks at.
        Vector3 ClearView(Vector3 subject, Vector3 look, Vector3 dir, float distance, float height, float clear)
        {
            foreach (float a in new[] { 0f, 30, -30, 55, -55, 80, -80, 120, -120, 180 })
            {
                var d = Quaternion.AngleAxis(a, Vector3.up) * dir; bool ok = true;
                for (float k = 0; k <= 1.001f && ok; k += .5f)
                {
                    var p = subject + d * distance * Mathf.Lerp(1, .85f, k); p.y = Mathf.Max(subject.y + height, G(p) + clear);
                    if (p.y - Mathf.Max(subject.y, G(p)) > height + 6) { ok = false; break; }   // lifted over a tree: try elsewhere
                    if (Physics.CheckSphere(p, .7f, ~0, QueryTriggerInteraction.Ignore) || NearSand(p, subject)) { ok = false; break; }
                    var to = look - p;
                    if (Physics.Raycast(p, to.normalized, to.magnitude - 2.5f, ~0, QueryTriggerInteraction.Ignore)) ok = false;
                }
                if (ok) return d;
            }
            return dir;
        }
        /// The lens over or beside another hazard's lip (bunker faces have no colliders): somewhere else.
        bool NearSand(Vector3 p, Vector3 subject)
        {
            for (int i = 0; i < 8; i++)
            {
                float a = i * Mathf.PI / 4; var q = p + new Vector3(Mathf.Cos(a), 0, Mathf.Sin(a)) * 5;
                var kind = hole.HazardAt(new CoursePoint(q.x, q.z));
                if (kind == HazardKind.Bunker && Vector3.Distance(new Vector3(q.x, 0, q.z), new Vector3(subject.x, 0, subject.z)) > 8) return true;
            }
            return hole.HazardAt(new CoursePoint(p.x, p.z)) == HazardKind.Bunker && Vector3.Distance(p, subject) > 8;
        }
        readonly Dictionary<Feature, Vector3> views = new();
        Vector3 Placed(Feature f, Vector3 look, Vector3 dir, float distance, float k, float height, float clear)
        {
            if (!views.TryGetValue(f, out var d)) views[f] = d = ClearView(f.at, look, dir, distance, height, clear);
            var p = f.at + d * distance * Mathf.Lerp(1, .85f, k); p.y = Mathf.Max(f.at.y + height, G(p) + clear);
            return p;
        }
        /// The shot for each kind of highlight: what makes it read, with the target beyond it where that matters.
        void FeatureShot(Feature f, float t, float len)
        {
            float k = S(t / len);
            var toPin = pin - f.at; toPin.y = 0; toPin = toPin.sqrMagnitude > 1 ? toPin.normalized : Fwd(.9f);
            var side = Vector3.Cross(Vector3.up, toPin);
            float onFairway = Mathf.Sign(Vector3.Dot(AtYards(AlongOf(f.at)) - f.at, side)); if (onFairway == 0) onFairway = 1;
            switch (f.kind)
            {
                case "carry":
                {
                    // skimming low over the water toward the far bank and the green on it
                    float from = AlongOf(f.at) - f.size / 2;
                    var p = Above(AtYards(from + f.size * Mathf.Lerp(-.15f, .25f, k)) + Right(.5f) * 10 + Vector3.up * 3.2f, 2.5f);
                    Cam(p, AtYards(from + f.size) + Vector3.up * 2, 40); NoFocus(); break;
                }
                case "windmill":
                {
                    var hub = f.at; float a = Mathf.Lerp(-20, 15, k) * Mathf.Deg2Rad; var back = -toPin;
                    Cam(hub + (back * Mathf.Cos(a) + side * Mathf.Sin(a)) * f.size * 3.2f + Vector3.up * .5f, hub, 38); NoFocus(); break;
                }
                case "greenside":
                {
                    // low behind the sand, the flag beyond it
                    var away = f.at - pin; away.y = 0; away = away.sqrMagnitude > 1 ? away.normalized : -toPin;
                    // (raised so the sand sits low in the frame and the green and its flag read beyond it)
                    var look = Vector3.Lerp(f.at, pin, .7f) + Vector3.up * .6f;
                    var p = Placed(f, look, (away + Vector3.Cross(Vector3.up, away) * .35f).normalized, 19, k, 5.5f, 3.5f);
                    Cam(p, look, 36); NoFocus(); break;
                }
                case "edge":
                {
                    // from out over the drop, the fairway on top of the cliff
                    var land = AtYards(AlongOf(f.at)); var outward = f.at - land; outward.y = 0; outward = outward.normalized;
                    var p = f.at + outward * Mathf.Lerp(34, 28, k) - toPin * 10 + Vector3.up * (G(land) + 5 - f.at.y);
                    Cam(p, land + toPin * 30 + Vector3.up * 1, 42); NoFocus(); break;
                }
                default:
                {
                    // fairway bunkers, pools, rocks: low on the fairway side, past them toward the green
                    var look = Vector3.Lerp(f.at, pin, .2f) + Vector3.up * .8f;
                    var p = Placed(f, look, (-toPin + side * onFairway * .45f).normalized, f.size * 1.1f + 16, k, 3.4f, 2.4f);
                    Cam(p, look, 40); NoFocus(); break;
                }
                case "green":
                {
                    // low, long lens: the flag sharp, the view behind it soft
                    var approach = -Fwd(.97f); var across = Vector3.Cross(Vector3.up, approach);
                    var look = pin + Vector3.up * .9f;
                    var p = Placed(f, look, (approach * .7f + across * -sea * .7f).normalized, 12, k, 2.1f, 1.6f);
                    Cam(p, look, 28); NoFocus(); break;   // (far blur against open sky streaks: the long lens alone)
                }
            }
            Handheld(t + f.score, .8f);
        }

                    }

            finally { Time.captureFramerate = oldRate; beforeCapture = null; if (canvas) Object.Destroy(canvas.gameObject); }
        }
